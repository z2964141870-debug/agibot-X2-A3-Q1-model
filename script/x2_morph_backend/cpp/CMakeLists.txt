cmake_minimum_required(VERSION 3.16)
project(x2_morph_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(aimdk_msgs REQUIRED)
find_package(Threads REQUIRED)
find_library(ZMQ_LIBRARY zmq REQUIRED)
set(SONIC_SOURCE "/home/agi/gear-sonic/ws/src/x2/agi_x2_deploy_onnx_ref" CACHE PATH "Existing Sonic source")
set(MORPH_RUNTIME "${CMAKE_CURRENT_LIST_DIR}/../../../data/runtime/x2_morph_human" CACHE PATH "Verified Morph runtime")
add_executable(x2_morph_native native_controller.cpp "${SONIC_SOURCE}/src/aimdk_io.cpp")
target_include_directories(x2_morph_native PRIVATE "${SONIC_SOURCE}/include" "${MORPH_RUNTIME}/cpp_deps")
target_link_libraries(x2_morph_native ${ZMQ_LIBRARY} Threads::Threads)
ament_target_dependencies(x2_morph_native rclcpp sensor_msgs aimdk_msgs)
target_compile_options(x2_morph_native PRIVATE -Wall -Wextra -Werror)
add_executable(test_morph_writer test_writer_gate.cpp)
target_compile_options(test_morph_writer PRIVATE -Wall -Wextra -Werror -UNDEBUG)
enable_testing()
add_test(NAME morph_writer_gate COMMAND test_morph_writer)
